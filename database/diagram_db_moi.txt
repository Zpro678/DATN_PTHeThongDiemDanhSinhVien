Enum user_role {
  USER          [note: 'Người dùng bình thường (Giảng viên/Sinh viên)']
  ADMIN         [note: 'Quản trị viên quản lý hệ thống']
  SUPER_ADMIN   [note: 'Quản trị viên tối cao có toàn quyền']
}

Table users {
  id bigint [pk, increment, note: 'Khóa chính tự tăng của người dùng']
  role varchar(50) [not null, default: 'USER', note: 'Vai trò phân quyền: USER, ADMIN, SUPER_ADMIN']
  google_id varchar [null, unique, note: 'ID tài khoản liên kết Google OAuth']
  name varchar [not null, note: 'Họ và tên người dùng']
  email varchar [unique, not null, note: 'Email đăng nhập duy nhất']
  email_verified_at timestamp [null, note: 'Mốc thời gian xác minh email']
  password varchar [null, note: 'Mật khẩu đã mã hóa (null nếu đăng nhập qua Google)']
  avatar varchar [null, note: 'Đường dẫn hoặc URL ảnh đại diện']
  status varchar(50) [not null, default: 'active', note: 'Trạng thái tài khoản: active | blocked']
  telegram_chat_id varchar [null, note: 'Chat ID kết nối nhận thông báo qua Telegram']
  notification_preferences json [null, note: 'Cấu hình nhận thông báo (mail, database, telegram)']
  remember_token varchar [null, note: 'Token ghi nhớ đăng nhập']
  created_at timestamp [note: 'Thời điểm tạo tài khoản']
  updated_at timestamp [note: 'Thời điểm cập nhật thông tin']
  deleted_at timestamp [null, note: 'Thời điểm xóa mềm tài khoản']
}

Table password_reset_tokens {
  email varchar [pk, note: 'Email yêu cầu đặt lại mật khẩu']
  token varchar [not null, note: 'Token xác thực đặt lại mật khẩu']
  created_at timestamp [null, note: 'Thời điểm tạo token']
}


Table sessions {
  id varchar [pk, note: 'ID phiên truy cập']
  user_id bigint [null, ref: > users.id, note: 'ID người dùng sở hữu phiên (nếu đã đăng nhập)']
  ip_address varchar(45) [null, note: 'Địa chỉ IP của thiết bị']
  user_agent text [null, note: 'Thông tin trình duyệt/thiết bị truy cập']
  payload longtext [not null, note: 'Dữ liệu phiên được mã hóa']
  last_activity int [not null, note: 'Thời điểm hoạt động cuối cùng (timestamp)']
}

// ======================================================
// 2. CLASSES & ENROLLMENT (LỚP HỌC & THÀNH VIÊN)

Table classes {
  id uuid [pk, note: 'Khóa chính UUID của lớp học']
  owner_user_id bigint [not null, ref: > users.id, note: 'ID của giảng viên sở hữu lớp chính (restrictOnDelete)']
  join_key varchar(50) [unique, not null, note: 'Mã tham gia lớp ngẫu nhiên cho học viên']
  class_code varchar(50) [null, note: 'Mã lớp do giảng viên tự đặt theo khoa/trường (VD: CS101)']
  name varchar [not null, note: 'Tên lớp học/môn học']
  description text [null, note: 'Mô tả chi tiết lớp học']
  late_threshold int [not null, default: 15, note: 'Số phút trễ tối đa cho phép để tính đi muộn']
  deduct_excused_absence boolean [not null, default: false, note: 'Có trừ chuyên cần khi vắng có phép hay không']
  require_approval boolean [not null, default: false, note: 'Bật/tắt yêu cầu duyệt khi xin vào lớp']
  status varchar(50) [not null, default: 'active', note: 'Trạng thái lớp học: active | archived']
  total_sessions int [not null, default: 15, note: 'Tổng số buổi học dự kiến của môn học']
  deduct_late decimal(4,2) [not null, default: 0.5, note: 'Điểm trừ khi đi muộn']
  deduct_absent decimal(4,2) [not null, default: 1.0, note: 'Điểm trừ khi vắng mặt']
  deduct_excused decimal(4,2) [not null, default: 0.0, note: 'Điểm trừ khi vắng có phép']
  created_at timestamp [note: 'Thời điểm tạo lớp']
  updated_at timestamp [note: 'Thời điểm sửa thông tin lớp']
  deleted_at timestamp [null, note: 'Thời điểm xóa mềm lớp học']
}

Table class_owners {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  class_id uuid [not null, ref: > classes.id, note: 'ID lớp học liên kết (cascadeOnDelete)']
  user_id bigint [not null, ref: > users.id, note: 'ID giảng viên đồng quản lý (cascadeOnDelete)']
  role varchar(50) [not null, default: 'co_owner', note: 'Vai trò đồng chủ quản lý lớp']
  invited_by bigint [null, ref: > users.id, note: 'ID giảng viên gửi lời mời (nullOnDelete)']
  accepted_at timestamp [null, note: 'Thời điểm chấp nhận lời mời làm đồng chủ lớp']
  created_at timestamp [note: 'Thời điểm gửi lời mời']
  updated_at timestamp [note: 'Thời điểm cập nhật trạng thái']
}

Enum member_status {
  ACTIVE     [note: 'Thành viên đang học bình thường']
  LEFT       [note: 'Thành viên tự rời khỏi lớp']
  REMOVED    [note: 'Thành viên bị giảng viên xóa khỏi lớp']
}

Table class_members {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  class_id uuid [not null, ref: > classes.id, note: 'ID lớp học (cascadeOnDelete)']
  user_id bigint [null, ref: > users.id, note: 'ID tài khoản liên kết sau khi sinh viên đăng ký (nullOnDelete)']
  status varchar(50) [not null, default: 'ACTIVE', note: 'Trạng thái: ACTIVE | LEFT | REMOVED']
  status_changed_at timestamp [null, note: 'Thời điểm thay đổi trạng thái thành viên']
  created_at timestamp [note: 'Thời điểm tham gia lớp']
  updated_at timestamp [note: 'Thời điểm cập nhật thông tin']
  deleted_at timestamp [null, note: 'Thời điểm xóa mềm thành viên']

  indexes {
    (class_id, user_id) [unique]
    user_id
  }
}

Table class_member_profiles {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  class_member_id bigint [unique, not null, ref: - class_members.id, note: 'ID thành viên liên kết 1-1 (cascadeOnDelete)']
  full_name varchar [null, note: 'Họ và tên hiển thị trong danh sách lớp']
  email varchar [null, note: 'Email liên hệ của sinh viên']
  created_at timestamp [note: 'Thời điểm tạo hồ sơ']
  updated_at timestamp [note: 'Thời điểm cập nhật hồ sơ']
}

Enum request_status {
  PENDING    [note: 'Đang chờ giảng viên duyệt vào lớp']

  REJECTED   [note: 'Đã từ chối yêu cầu tham gia']

Table class_join_requests {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  class_id uuid [not null, ref: > classes.id, note: 'ID lớp học muốn tham gia']
  user_id bigint [not null, ref: > users.id, note: 'ID tài khoản sinh viên gửi yêu cầu'] 
  status varchar(50) [not null, default: 'PENDING', note: 'Trạng thái yêu cầu: PENDING | APPROVED | REJECTED']
  created_at timestamp [note: 'Thời điểm gửi yêu cầu']
  updated_at timestamp [note: 'Thời điểm duyệt/từ chối']
  indexes { (class_id, status) }
}

Table class_sessions {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  meeting_id bigint [null, ref: > class_meetings.id, note: 'ID buổi học cha chứa phiên (cascadeOnDelete)']
  class_id uuid [not null, ref: > classes.id, note: 'ID lớp học chứa phiên (cascadeOnDelete)']
  created_by bigint [not null, ref: > users.id, note: 'ID người tạo phiên điểm danh (restrictOnDelete)']
  name varchar [not null, note: 'Tên phiên điểm danh (VD: Phiên 1, Điểm danh đầu giờ)']
  date date [not null, note: 'Ngày diễn ra điểm danh']
  start_time time [null, note: 'Giờ bắt đầu cho phép điểm danh']
  end_time time [null, note: 'Giờ kết thúc chốt điểm danh']
  qr_token varchar [unique, null, note: 'Token mã hóa của QR chống gian lận']
  token_expires_at timestamp [null, note: 'Thời điểm mã QR hiện tại hết hạn']
  qr_refresh_rate int [not null, default: 10, note: 'Thời gian tự động làm mới mã QR (tính bằng giây)']
  gps_latitude decimal(10,8) [null, note: 'Kinh độ GPS mốc của giảng viên']
  gps_longitude decimal(11,8) [null, note: 'Vĩ độ GPS mốc của giảng viên']
  gps_radius int [null, note: 'Bán kính sai số GPS cho phép (mét)']
  device_check boolean [not null, default: false, note: 'Bật/tắt kiểm tra trùng thiết bị điểm danh']
  status varchar(50) [not null, default: 'pending', note: 'Trạng thái phiên: pending | active | closed']
  share_token varchar(64) [unique, null, note: 'Token dùng để sinh link điểm danh nhanh']
  created_at timestamp [note: 'Thời điểm tạo phiên']
  updated_at timestamp [note: 'Thời điểm cập nhật phiên']
  deleted_at timestamp [null, note: 'Thời điểm xóa mềm phiên']

  indexes {
    (class_id, date)
    (class_id, created_at)
    (qr_token, token_expires_at)
  }
}

Table attendance_records {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  class_session_id bigint [not null, ref: > class_sessions.id, note: 'ID phiên điểm danh liên kết (cascadeOnDelete)']
  class_member_id bigint [not null, ref: > class_members.id, note: 'ID thành viên lớp điểm danh (cascadeOnDelete)']
  status varchar(50) [not null, default: 'pending', note: 'Trạng thái điểm danh: pending | present | late | absent | excused | invalid']
  is_account boolean [not null, default: false, note: 'Xác định sinh viên tự điểm danh qua tài khoản (true) hay điền form/giảng viên đánh dấu (false)']
  check_in_time timestamp [null, note: 'Thời điểm quét điểm danh thành công']
  ip_address varchar(45) [null, note: 'Địa chỉ IP thiết bị của sinh viên khi check-in']
  device_id varchar [null, note: 'ID định danh thiết bị vật lý']
  device_fingerprint varchar [null, note: 'Dấu vân tay trình duyệt của thiết bị']
  distance_meters decimal(8,2) [null, note: 'Khoảng cách thực tế giữa sinh viên và giảng viên (mét)']
  gps_accuracy_meters decimal(8,2) [null, note: 'Độ chính xác đo đạc thiết bị sinh viên báo về']
  gps_latitude_recorded decimal(10,8) [null, note: 'Kinh độ GPS thực tế của sinh viên khi quét']
  gps_longitude_recorded decimal(11,8) [null, note: 'Vĩ độ GPS thực tế của sinh viên khi quét']
  gps_fraud_flag varchar [null, note: 'Cờ cảnh báo nếu phát hiện nghi ngờ giả lập/fake tọa độ GPS']
  note text [null, note: 'Ghi chú thêm về bản ghi']
  created_at timestamp [note: 'Thời điểm tạo bản ghi']
  updated_at timestamp [note: 'Thời điểm sửa bản ghi']
  deleted_at timestamp [null, note: 'Thời điểm xóa mềm bản ghi']

  indexes {
    (class_session_id, class_member_id) [unique]
    (class_session_id, status)
  }
}

Table check_in_scans {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  class_session_id bigint [not null, ref: > class_sessions.id, note: 'ID phiên điểm danh quét quét (cascadeOnDelete)']

  is_valid boolean [not null, default: true, note: 'Kết quả quét có hợp lệ theo các quy tắc bảo mật không']
  fail_reason varchar(100) [null, note: 'Lý do quét thất bại: timeout | invalid_signature | out_of_range | not_enrolled']
  ip_address varchar(45) [null, note: 'Địa chỉ IP của thiết bị check-in']
  device_id varchar [null, note: 'ID phần cứng thiết bị check-in']
  device_fingerprint varchar [null, note: 'Dấu vân tay trình duyệt check-in']
  scanned_at timestamp [not null, note: 'Thời điểm hệ thống nhận tín hiệu quét']

  indexes {
    (class_session_id, scanned_at)
    (user_id, scanned_at)
    (device_fingerprint, scanned_at)
  }
}

  updated_at timestamp [note: 'Thời điểm cập nhật']

Table transactions {
  id bigint [pk, increment, note: 'Khóa chính tự tăng']
  user_id bigint [not null, ref: > users.id, note: 'ID người thực hiện thanh toán giao dịch (restrictOnDelete)']
  plan_id bigint [null, ref: > plans.id, note: 'ID gói đăng ký được thanh toán (nullOnDelete)']
  coupon_id bigint [null, ref: > coupons.id, note: 'ID mã giảm giá áp dụng (nếu có)']
  coupon_code varchar(50) [null, note: 'Lưu text mã giảm giá phòng khi coupon bị xóa khỏi hệ thống']
  amount decimal(10,2) [not null, note: 'Số tiền thực tế thanh toán sau giảm giá']
  currency varchar(10) [not null, default: 'VND', note: 'Đơn vị tiền tệ thanh toán']
  payment_method varchar(50) [not null, note: 'Phương thức thanh toán: PayOS, VNPay, MoMo...']
  transaction_code varchar(100) [unique, not null, note: 'Mã giao dịch nội bộ duy nhất']
  reference_code varchar(100) [unique, null, note: 'Mã đối chiếu từ hệ thống thanh toán bên thứ ba']
  gateway_transaction_id varchar(100) [null, note: 'Mã định danh giao dịch của ngân hàng/cổng thanh toán']
  status varchar(50) [not null, default: 'PENDING', note: 'Trạng thái giao dịch: PENDING | SUCCESS | FAILED | EXPIRED']
  payment_url text [null, note: 'Đường dẫn liên kết thanh toán bên đối tác']
  payment_response json [null, note: 'Dữ liệu thô phản hồi từ Cổng thanh toán (Webhook/IPN)']
  failure_reason text [null, note: 'Lý do giao dịch bị thất bại nếu có']
  paid_at timestamp [null, note: 'Mốc thời gian giao dịch được xác nhận thanh toán thành công']
  expired_at timestamp [null, note: 'Mốc thời gian giao dịch bị hết hạn thanh toán']
  created_at timestamp [null, note: 'Thời điểm bắt đầu tạo giao dịch']
  updated_at timestamp [null, note: 'Thời điểm cập nhật trạng thái']
  
  indexes {
    (user_id, created_at)
    (transaction_code)
    (reference_code)
    (gateway_transaction_id)
    (status)
  }
}
